#include <ZMPT101B.h> // For ZMPT101B voltage sensor
#include "ACS712.h" // For ACS712 current sensor
#include <SoftwareSerial.h> // For communication with ESP32

// Define sensor pins
const int voltagePin = A0;    // ZMPT101B connected to A0
const int currentPin = A2;    // ACS712 connected to A2

// Create sensor objects
ZMPT101B voltageSensor(voltagePin, 5.0); // (pin, Vref)
ACS712 currentSensor(ACS712_20A, currentPin); // Change ACS712_20A to match your sensor rating

// UART pins for communication with ESP32
SoftwareSerial espSerial(2, 3); // RX, TX

// Define relay pins
#define RELAY1_PIN 4  // Relay 1 connected to digital pin 4
#define RELAY2_PIN 5  // Relay 2 connected to digital pin 5

void setup() {
  Serial.begin(115200); // Serial monitor for debugging
  espSerial.begin(115200); // Communication with ESP32

  // Configure relay pins as output
  pinMode(RELAY1_PIN, OUTPUT);
  pinMode(RELAY2_PIN, OUTPUT);

  // Initialize relays to OFF state
  digitalWrite(RELAY1_PIN, HIGH); // Relays are active LOW
  digitalWrite(RELAY2_PIN, HIGH);

  // Calibrate current sensor
  currentSensor.calibrate();

  Serial.println("Arduino ready to measure voltage, current, and control relays.");
}

void loop() {
  // Measure AC voltage using ZMPT101B
  float voltage = voltageSensor.getRmsVoltage();
  voltage = (voltage < 150) ? 0 : voltage; // Protect against low voltage readings

  // Measure AC current using ACS712
  float current = currentSensor.getCurrentAC();
  if (current < 0.09) current = 0; // Ignore noise values below 0.09A

  // Log the data for debugging
  Serial.print("Voltage: ");
  Serial.print(voltage, 2);
  Serial.print(" V, Current: ");
  Serial.print(current, 2);
  Serial.println(" A");

  // Send data to ESP32
  espSerial.print("Voltage: ");
  espSerial.print(voltage, 2);
  espSerial.print(" V, Current: ");
  espSerial.print(current, 2);
  espSerial.println(" A");

  // Wait for acknowledgment from ESP32
  if (espSerial.available() > 0) {
    String ack = espSerial.readStringUntil('\n');
    if (ack == "ACK") {
      Serial.println("Acknowledgment received from ESP32.");
    } else {
      Serial.println("Failed to receive valid acknowledgment from ESP32.");
    }
  }

  // Check if data is available from ESP32 for relay control
  if (Serial.available()) {
    String command = Serial.readStringUntil('\n');
    command.trim(); // Remove any extra spaces or newline characters

    Serial.print("Received command: ");
    Serial.println(command);

    // Process relay commands
    if (command == "ON1") {
      digitalWrite(RELAY1_PIN, LOW); // Turn ON Relay 1
      Serial.println("Relay 1 turned ON.");
    } else if (command == "OFF1") {
      digitalWrite(RELAY1_PIN, HIGH); // Turn OFF Relay 1
      Serial.println("Relay 1 turned OFF.");
    } else if (command == "ON2") {
      digitalWrite(RELAY2_PIN, LOW); // Turn ON Relay 2
      Serial.println("Relay 2 turned ON.");
    } else if (command == "OFF2") {
      digitalWrite(RELAY2_PIN, HIGH); // Turn OFF Relay 2
      Serial.println("Relay 2 turned OFF.");
    } else {
      Serial.println("Invalid command received.");
    }
  }

  delay(1000); // Wait for 1 second
}
