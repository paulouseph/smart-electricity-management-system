const char* password[] = {"123456789", "george7308", "niMItm@Wifi", "12345678", "chubby82"};
const int numberOfNetworks = 5;

// Firebase configuration
FirebaseData firebaseData;
FirebaseConfig firebaseConfig;
FirebaseAuth firebaseAuth;

const char* firebaseHost = "https://sem-system-93f17-default-rtdb.asia-southeast1.firebasedatabase.app/";

// Define RX and TX pins for Arduino communication
#define RX_PIN 16  // RX2 pin for ESP32
#define TX_PIN 17  // TX2 pin for ESP32

// Variables for voltage and current
float voltage = 0.0;
float current1 = 0.0;
float current2 = 0.0;
float appliance1KW = 0.0;
float appliance2KW = 0.0;

// Variables to track Firebase relay states
String desiredRelay1State = "OFF";
String desiredRelay2State = "OFF";
String lastRelay1State = "OFF";
String lastRelay2State = "OFF";

// Timer for Firebase update every 40 seconds
unsigned long lastFirebaseUpdate = 0;
const unsigned long firebaseUpdateInterval = 40000; // 40 seconds

// Function to connect to Wi-Fi
void connectToWiFi() {
  for (int i = 0; i < numberOfNetworks; i++) {
    WiFi.begin(ssid[i], password[i]);
    Serial.print("Connecting to WiFi: ");
    Serial.println(ssid[i]);

    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
      delay(500);
      Serial.print(".");
      attempts++;
    }

    if (WiFi.status() == WL_CONNECTED) {
      Serial.println("\nConnected to WiFi");
      Serial.print("IP Address: ");
      Serial.println(WiFi.localIP());
      return;
    }
    Serial.println("\nFailed to connect. Trying next network...");
  }

  Serial.println("Failed to connect to all networks. Restarting...");
  ESP.restart();
}

// Send relay command to Arduino
void sendRelayCommand(String command) {
  Serial2.println(command); // Send command to Arduino
  Serial.print("Sent command to Arduino: ");
  Serial.println(command);
}

void setup() {
  Serial.begin(115200); // Debugging at 115200 baud rate
  Serial2.begin(115200, SERIAL_8N1, RX_PIN, TX_PIN); // Communication with Arduino

  connectToWiFi();

  // Configure Firebase
  firebaseConfig.host = firebaseHost;
  firebaseConfig.signer.tokens.legacy_token = firebaseSecret;

  Firebase.begin(&firebaseConfig, &firebaseAuth);
  Firebase.reconnectWiFi(true);

  Serial.println("Firebase initialized.");
}

void loop() {
  // Read data from Arduino
  if (Serial2.available()) {
    String data = Serial2.readStringUntil('\n');
    data.trim();

    Serial.print("Data from Arduino: ");
    Serial.println(data);

    // Parse data for voltage and currents
    int voltageIndex = data.indexOf("Voltage: ");
    int appliance1Index = data.indexOf("Appliance 1 Current: ");
    int appliance2Index = data.indexOf("Appliance 2 Current: ");

    if (voltageIndex != -1 && appliance1Index != -1 && appliance2Index != -1) {
      voltage = data.substring(voltageIndex + 9, data.indexOf(" V", voltageIndex)).toFloat();
      current1 = data.substring(appliance1Index + 21, data.indexOf(" A", appliance1Index)).toFloat();
      current2 = data.substring(appliance2Index + 21, data.indexOf(" A", appliance2Index)).toFloat();

      // Calculate power (in kW) based on current and voltage
      appliance1KW = (current1 * voltage) / 1000.0; // Power in kW for appliance 1
      appliance2KW = (current2 * voltage) / 1000.0; // Power in kW for appliance 2

      // If relay is OFF, set power to 0
      if (desiredRelay1State == "OFF") {
        appliance1KW = 0.0;
      }

      if (desiredRelay2State == "OFF") {
        appliance2KW = 0.0;
      }

      // Print calculated values
      Serial.print("Appliance 1 Power: ");
      Serial.print(appliance1KW, 4);
      Serial.println(" kW");

      Serial.print("Appliance 2 Power: ");
      Serial.print(appliance2KW, 4);
      Serial.println(" kW");
    }
  }

  // Check Firebase for desired relay states
  if (Firebase.getString(firebaseData, "/Appliance1/DesiredRelayState")) {
    desiredRelay1State = firebaseData.stringData();
  }
  if (Firebase.getString(firebaseData, "/Appliance2/DesiredRelayState")) {
    desiredRelay2State = firebaseData.stringData();
  }

  // Send relay commands if desired states have changed
  if (desiredRelay1State != lastRelay1State) {
    sendRelayCommand(desiredRelay1State == "ON" ? "ON1" : "OFF1");
    lastRelay1State = desiredRelay1State;
  }

  if (desiredRelay2State != lastRelay2State) {
    sendRelayCommand(desiredRelay2State == "ON" ? "ON2" : "OFF2");
    lastRelay2State = desiredRelay2State;
  }

  // Update Firebase every 40 seconds
  unsigned long currentMillis = millis();
  if (currentMillis - lastFirebaseUpdate >= firebaseUpdateInterval) {
    lastFirebaseUpdate = currentMillis;

    // Update Firebase with appliance power values
    Firebase.setFloat(firebaseData, "/Appliance1/Power", appliance1KW);
    Firebase.setFloat(firebaseData, "/Appliance2/Power", appliance2KW);
  }

  delay(100); // Polling delay for checking relay states and updating power
}
